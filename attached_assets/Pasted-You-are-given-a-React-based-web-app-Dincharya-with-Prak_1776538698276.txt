PART 4: GEMINI FUNCTION (WITH POPUP)

async function getDietPlan() {
  try {
    const res = await fetch(
      \`https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=\${import.meta.env.VITE_GEMINI_API_KEY}\`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          system_instruction: { parts: [{ text: systemInstruction }] },
          contents: [{ parts: [{ text: userPrompt }] }]
        })
      }
    );

    const data = await res.json();

    let text = data.candidates[0].content.parts[0].text;
    text = text.replace(/```json|```/g, "").trim();

    alert("✅ AI Diet Plan Generated");

    return JSON.parse(text);

  } catch (err) {
    alert("❌ Failed to generate diet plan");
    console.error(err);
  }
}

PART 5: BACKEND CALORIE API (CalorieNinjas)

Create Express route:

app.get("/api/nutrition", async (req, res) => {
  try {
    const query = req.query.food;

    const response = await fetch(
      \`https://api.calorieninjas.com/v1/nutrition?query=\${query}\`,
      {
        method: "GET",
        headers: {
          "X-Api-Key": process.env.CALORIE_NINJA_API_KEY
        }
      }
    );

    const data = await response.json();

    res.json(data);

  } catch (err) {
    res.status(500).json({ error: "Failed to fetch nutrition" });
  }
});
